## Binning
input_directory <- paste0(wd, ".../exported_data/")  # Update this path
output_directory <- paste0(wd, ".../binned_data/")     # Update this path
group_size <- 6 # change based on elements in the group -> ppm setting of the file

csv_files <- list.files(input_directory, pattern = "*.csv", full.names = TRUE)

# Main processing loop
for (file in csv_files) {
  
  # ---- Load and Parse Data ----
  startdata <- read_delim(file, delim = ";", escape_double = FALSE, trim_ws = TRUE, skip = 10)
  dataframe <- startdata[, -1]
  mz_values <- as.numeric(gsub("mz_", "", colnames(dataframe)))
  mz_diff <- diff(mz_values)
  mz_diff[is.na(mz_diff)] <- 0
  
  # ---- Dynamic Thresholding ----
  start_thresh <- mz_diff[1] * 1.5
  end_thresh <- mz_diff[length(mz_diff)] * 1.5
  thresholds <- seq(start_thresh, end_thresh, length.out = length(mz_diff))
  
  # ---- Group Columns by m/z Similarity ----
  group_id <- 1
  group_labels <- numeric(length(mz_values))
  group_labels[1] <- group_id
  
  for (i in 1:length(mz_diff)) {
    if (mz_diff[i] > thresholds[i]) {
      group_id <- group_id + 1
    }
    group_labels[i + 1] <- group_id
  }
  
  grouped_columns <- split(seq_along(mz_values), group_labels)
  
  # ---- Refine Large Groups (optional) ----
  split_group <- function(group, size) {
    n_sub <- ceiling(length(group) / size)
    split(group, rep(1:n_sub, each = size, length.out = length(group)))
  }
  
  grouped_columns <- unlist(lapply(grouped_columns, function(g) {
    if (length(g) > group_size) split_group(g, group_size) else list(g)
  }), recursive = FALSE)
  
  # ---- Summarize by Group ----
  group_averages <- sapply(grouped_columns, function(group) mean(mz_values[group], na.rm = TRUE))
  max_per_group <- lapply(grouped_columns, function(group) {
    apply(dataframe[, group, drop = FALSE], 1, max, na.rm = TRUE)
  })
  
  max_df <- as.data.frame(do.call(cbind, max_per_group))
  colnames(max_df) <- paste0("X", round(group_averages, 4))
  rownames(max_df) <- startdata[[1]]
  
  # ---- Write Output ----
  output_file <- file.path(output_directory, paste0(tools::file_path_sans_ext(basename(file)), "_binned.csv"))
  write.csv(max_df, output_file, row.names = TRUE)
}

# Remove lipids with NA > 20%
# Include line which prints the lipid features that are removed
drop_features_with_high_na <- function(df, threshold = 0.2, start_col = 6) {
  # Split the data
  preserved_cols <- df[, 1:(start_col - 1)]
  target_cols <- df[, start_col:ncol(df)]
  
  # Compute NA percentages
  na_percentage <- colMeans(is.na(target_cols))
  
  # Identify columns to remove
  removed_cols <- names(na_percentage[na_percentage > threshold])
  if (length(removed_cols) > 0) {
    cat("Removed columns due to high NA percentage:\n")
    print(removed_cols)
  } else {
    cat("No columns removed. All columns meet the NA threshold.\n")
  }
  
  # Filter and recombine
  filtered_target <- target_cols[, na_percentage <= threshold]
  filtered_df <- cbind(preserved_cols, filtered_target)
  return(filtered_df)
}

### Imputation: if data is missing, allocate average of either 1st or 2nd order neighbors
impute_missing_values_fast <- function(coords, data, step_size = 20) {
  coords_xy <- coords[, c("x", "y")]
  data_mat <- as.matrix(data[, 6:ncol(data)])
  
  # Precompute overall means once
  col_means <- colMeans(data_mat, na.rm = TRUE)
  
  # Create lookup key
  coord_keys <- interaction(coords_xy$x, coords_xy$y, drop = TRUE)
  coord_to_index <- setNames(seq_len(nrow(coords_xy)), coord_keys)
  
  # Offsets
  offsets_8 <- expand.grid(dx = c(-1, 0, 1), dy = c(-1, 0, 1)) %>%
    filter(!(dx == 0 & dy == 0)) %>%
    mutate(dx = dx * step_size, dy = dy * step_size)
  
  offsets_24 <- expand.grid(dx = -2:2, dy = -2:2) %>%
    filter(!(dx == 0 & dy == 0)) %>%
    mutate(dx = dx * step_size, dy = dy * step_size)
  
  # Status matrix
  status_mat <- matrix("original", nrow = nrow(data_mat), ncol = ncol(data_mat),
                       dimnames = list(rownames(data), colnames(data)[6:ncol(data)]))
  
  count_8 <- 0
  count_24 <- 0
  count_mean <- 0
  
  for (col_idx in seq_len(ncol(data_mat))) {
    missing_idx <- which(is.na(data_mat[, col_idx]))
    
    for (idx in missing_idx) {
      x <- coords_xy$x[idx]
      y <- coords_xy$y[idx]
      
      # Try 8 neighbors
      neighbor_coords_8 <- paste(x + offsets_8$dx, y + offsets_8$dy, sep = ".")
      idx_8 <- coord_to_index[neighbor_coords_8[neighbor_coords_8 %in% names(coord_to_index)]]
      vals_8 <- data_mat[idx_8, col_idx]
      vals_8 <- vals_8[!is.na(vals_8)]
      
      if (length(vals_8) > 0) {
        data_mat[idx, col_idx] <- mean(vals_8)
        status_mat[idx, col_idx] <- "imputed"
        count_8 <- count_8 + 1
        next
      }
      
      # Try 24 neighbors
      neighbor_coords_24 <- paste(x + offsets_24$dx, y + offsets_24$dy, sep = ".")
      idx_24 <- coord_to_index[neighbor_coords_24[neighbor_coords_24 %in% names(coord_to_index)]]
      vals_24 <- data_mat[idx_24, col_idx]
      vals_24 <- vals_24[!is.na(vals_24)]
      
      if (length(vals_24) > 0) {
        data_mat[idx, col_idx] <- mean(vals_24)
        status_mat[idx, col_idx] <- "imputed"
        count_24 <- count_24 + 1
      } else {
        data_mat[idx, col_idx] <- col_means[col_idx]
        status_mat[idx, col_idx] <- "imputed"
        count_mean <- count_mean + 1
      }
    }
  }
  
  imputed_df <- as.data.frame(data_mat)
  colnames(imputed_df) <- colnames(data)[6:ncol(data)]
  rownames(imputed_df) <- rownames(data)
  
  status_df <- as.data.frame(status_mat)
  
  cat("NaN transformed to numbers using the mean of 8 neighbors:", count_8, "\n")
  cat("NaN transformed to numbers using the mean of 24 neighbors:", count_24, "\n")
  cat("NaN transformed to the overall mean of the metabolite:", count_mean, "\n")
  
  return(list(imputed_data = imputed_df, status_df = status_df))
}

plot_metabolite_intensity <- function(data, metab, label, overlay_points) {
  
  subdata <- data
  subdata$intensity <- subdata[[metab]]
  
  q99 <- quantile(subdata$intensity, 0.99, na.rm = TRUE)
  subdata$intensity[subdata$intensity > q99] <- q99
  
  subdata$intensity <- (subdata$intensity - min(subdata$intensity, na.rm = TRUE)) /
    (max(subdata$intensity, na.rm = TRUE) - min(subdata$intensity, na.rm = TRUE))
  
  ggplot(subdata, aes(x = x, y = y)) +
    geom_tile(aes(fill = intensity)) +
    
    geom_point(
      data = overlay_points,
      aes(x = x, y = y),
      color = "pink",
      size = 4
    ) +
    
    scale_fill_viridis_c(limits = c(0, 1)) +
    coord_fixed() +
    theme_minimal() +
    theme(
      axis.text = element_blank(),
      axis.title = element_blank(),
      axis.ticks = element_blank(),
      plot.title = element_text(hjust = 0.5)
    ) +
    labs(title = label, fill = "Relative intensity")
}

# Method: ComBat batch correction
# Create batch vector from your metadata
batch <- metadata$batch  #Replace with your actual batch column name

# Create model matrix for biological covariates you want to preserve, in this case group (difference between AKI and CTRL)
# Adding celltype_label as a covariate did not make much of a difference in final cell population distinguishment
# Replace with your actual column names
mod <- model.matrix(~ group, data = metadata)

# Apply ComBat
combat_corrected <- ComBat(dat = lipid_matrix, 
                           batch = batch, 
                           mod = mod,
                           par.prior = T, #Uses parametric adjustments, which is more stable with smaller sample sizes and zero inflated data
                           prior.plots = FALSE, 
                           mean.only = F) #Corrects for both mean and variance batch effects, default ComBat setting, gave better results when comparing

# Transpose back to original format (pixels as rows, features as columns)

# Function to calculate morans i by spatial coordinates
calculate_morans_i_coords <- function(lipid_values, x_coords, y_coords, k_neighbors = 8) {
  # Create spatial points
  coords <- cbind(x_coords, y_coords)
  
  # Remove any NA values
  complete_idx <- complete.cases(cbind(lipid_values, coords))
  if (sum(complete_idx) < 10) {
    return(list(morans_i = NA, p_value = NA, expected = NA, variance = NA))
  }
  
  coords_clean <- coords[complete_idx, ]
  values_clean <- lipid_values[complete_idx]
  
  # Create k-nearest neighbors
  knn <- knearneigh(coords_clean, k = k_neighbors)
  nb <- knn2nb(knn)
  
  # Create spatial weights (row-standardized)
  weights <- nb2listw(nb, style = "W", zero.policy = TRUE)
  
  # Calculate Moran's I
  moran_result <- tryCatch({
    moran.test(values_clean, weights, zero.policy = TRUE)
  }, error = function(e) {
    return(list(estimate = c(NA, NA, NA), p.value = NA))
  })
  
  return(list(
    morans_i = moran_result$estimate[1],
    expected = moran_result$estimate[2], 
    variance = moran_result$estimate[3],
    p_value = moran_result$p.value
  ))
}

# Function to calculate spatial autocorr for top lipids by sample
calculate_spatial_autocorr_by_sample <- function(data, top_lipids, sample_col = "sample", 
                                                 x_col = "x", y_col = "y", k_neighbors = 8) {
  
  results_list <- list()
  samples <- unique(data[[sample_col]])
  
  cat("Calculating spatial autocorrelation for", length(top_lipids), "top lipids across", 
      length(samples), "samples...\n")
  
  for (sample_id in samples) {
    cat("Processing sample:", sample_id, "\n")
    
    # Subset data for this sample
    sample_data <- data[data[[sample_col]] == sample_id, ]
    
    # Check if we have enough points
    if (nrow(sample_data) < 10) {
      cat("  Warning: Sample", sample_id, "has <10 pixels, skipping\n")
      next
    }
    
    # Get coordinates for this sample
    x_coords <- sample_data[[x_col]]
    y_coords <- sample_data[[y_col]]
    
    for (lipid_name in top_lipids) {
      lipid_values <- sample_data[[lipid_name]]
      
      # Calculate Moran's I for this lipid in this sample
      result <- calculate_morans_i_coords(lipid_values, x_coords, y_coords, k_neighbors)
      
      # Store result with sample info
      results_list[[paste(sample_id, lipid_name, sep = "_")]] <- data.frame(
        Sample = sample_id,
        Lipid = lipid_name,
        Morans_I = result$morans_i,
        Expected_I = result$expected,
        Variance_I = result$variance,
        p_value_spatial = result$p_value,
        Z_score = ifelse(is.na(result$variance) || result$variance <= 0, NA, 
                         (result$morans_i - result$expected) / sqrt(result$variance)),
        n_pixels = length(lipid_values),
        stringsAsFactors = FALSE
      )
    }
  }
  
  # Combine all results
  spatial_results <- do.call(rbind, results_list)
  rownames(spatial_results) <- NULL
  
  # Add FDR correction within each sample
  spatial_results <- spatial_results %>%
    group_by(Sample) %>%
    mutate(p_value_spatial_fdr = p.adjust(p_value_spatial, method = "fdr")) %>%
    ungroup()
  
  return(spatial_results)
}

# Function to summarize spatial autocorrelation across samples
summarize_spatial_autocorr <- function(spatial_results) {
  summary_stats <- spatial_results %>%
    filter(!is.na(Morans_I)) %>%
    group_by(Lipid) %>%
    summarise(
      n_samples = n(),
      mean_morans_i = mean(Morans_I, na.rm = TRUE),
      median_morans_i = median(Morans_I, na.rm = TRUE),
      sd_morans_i = sd(Morans_I, na.rm = TRUE),
      min_morans_i = min(Morans_I, na.rm = TRUE),
      max_morans_i = max(Morans_I, na.rm = TRUE),
      prop_significant = mean(p_value_spatial_fdr < 0.05, na.rm = TRUE),
      prop_positive = mean(Morans_I > 0, na.rm = TRUE),
      prop_strong_positive = mean(Morans_I > 0.3, na.rm = TRUE),
      .groups = 'drop'
    ) %>%
    arrange(desc(mean_morans_i))
  
  return(summary_stats)
}

# Function to create a boxplot for Moran's I 
plot_morans_violin <- function(spatial_results, class_colors, manual_order = NULL) {
  
  # Add lipid class from name (before first ".")
  spatial_results <- spatial_results %>%
    filter(!is.na(Morans_I)) %>%
    mutate(Class = str_extract(Lipid, "^[^\\.]+"))
  
  # Assign base color per class
  base_pal <- function(class) class_colors[class]
  
  # Create lipid-specific palette (lighter shades for each lipid in a class)
  lipid_palette <- spatial_results %>%
    distinct(Lipid, Class) %>%
    group_by(Class) %>%
    mutate(
      n = n(),
      shade_id = row_number(),
      Color = lighten(base_pal(Class), amount = (shade_id - 1) / (n * 2))
    ) %>%
    ungroup() %>%
    select(Lipid, Color)
  
  # Join back
  spatial_results <- spatial_results %>%
    left_join(lipid_palette, by = "Lipid")
  
  # Use manual order if provided, otherwise order by median Moran's I
  if (!is.null(manual_order)) {
    spatial_results <- spatial_results %>%
      mutate(Lipid = factor(Lipid, levels = manual_order))
  } else {
    spatial_results <- spatial_results %>%
      mutate(Lipid = fct_reorder(Lipid, Morans_I, .fun = median))
  }
  
  ggplot(spatial_results, aes(x = Lipid, y = Morans_I, fill = Lipid)) +
    geom_violin(trim = FALSE, alpha = 0.8, scale = "width") +
    geom_boxplot(width = 0.1, outlier.size = 0.5, alpha = 0.4) +
    #geom_hline(yintercept = 0, linetype = "dotted", color = "black") +
    geom_hline(yintercept = 0.3, linetype = "dashed", color = "red") +
    scale_fill_manual(values = setNames(lipid_palette$Color, lipid_palette$Lipid)) +
    theme_minimal() +
    theme(axis.text.x = element_text(angle = 90, hjust = 0.5),
          legend.position = "none") +
    labs(
      title = "Distribution of Moran's I by Lipid",
      subtitle = "Colored by lipid class (different shades per lipid)",
      x = "Lipid",
      y = "Moran's I"
    )
}

create_spatial_visualization <- function(NEDC_segment_DE, NEDC_data_combat, 
                                         lipid_to_plot, sample_to_plot, 
                                         min_val, max_val, cap_global,
                                         custom_colors = NULL) {
  # Subset sample
  subdata <- NEDC_segment_DE %>%
    filter(sample == sample_to_plot)
  
  # Normalize using GLOBAL values
  if (max_val - min_val == 0) {
    subdata$normalized_percentage <- 0
  } else {
    subdata$normalized_percentage <- ((subdata[[lipid_to_plot]] - min_val) / 
                                        (max_val - min_val)) * 100
  }
  
  subdata$normalized_percentage <- pmin(subdata$normalized_percentage, cap_global)
  
  # Non-limma overlay
  df_nonprox <- NEDC_data_combat %>%
    filter(sample == sample_to_plot) %>%
    filter(!celltype_label %in% unique(NEDC_segment_DE$celltype_label))
  
  # Base plot with fixed limits
  p <- ggplot(subdata, aes(x = x, y = y, fill = normalized_percentage)) +
    geom_tile() +
    scale_fill_viridis_c(
      option = "viridis", 
      name = "Intensity (%)", 
      limits = c(0, cap_global)
    ) +
    coord_fixed() +
    theme_minimal() +
    theme(
      panel.grid = element_blank(),
      panel.background = element_rect(fill = "black", color = NA),
      plot.background = element_rect(fill = "black", color = NA),
      text = element_text(color = "white"),
      axis.text = element_blank(),
      axis.title = element_text(color = "white"),
      plot.title = element_text(color = "white")
    )
  
  if (nrow(df_nonprox) > 0) {
    p <- p + geom_point(
      data = df_nonprox,
      aes(x = x, y = y, colour = celltype_label),
      inherit.aes = FALSE,
      size = 1.0, alpha = 0.1
    ) +
      theme(legend.position = "right")
  }
  
  if (!is.null(custom_colors)) {
    p <- p + scale_color_manual(values = custom_colors, name = "Other clusters")
  }
  
  return(p)
}

## ---- Making sure that the colnames of all the datafiles are exactly the same as the colnames from the other dataset

# Function to match these colnames
harmonize_to_reference <- function(data_matrix, reference_names, tolerance = 1e-3) {
  # Strip "X" and cast to numeric
  curr_numeric <- as.numeric(gsub("^X", "", colnames(data_matrix)))
  ref_numeric  <- as.numeric(gsub("^X", "", reference_names))
  
  # For each reference mz, find closest in current
  matched_idx <- sapply(ref_numeric, function(ref_mz) {
    diffs <- abs(curr_numeric - ref_mz)
    if (min(diffs) <= tolerance) {
      return(which.min(diffs))
    } else {
      return(NA)
    }
  })
  
  # Reorder to reference order
  harmonized <- data_matrix[, matched_idx, drop = FALSE]
  
  # Assign *exactly* reference colnames
  colnames(harmonized) <- reference_names
  
  return(harmonized)
}

run_paired_limma <- function(data, condition_name) {
  
  # Filter for one condition and only T6 + T24
  df <- data %>%
    filter(Condition == condition_name,
           Timepoint %in% c("T6", "T24"))
  
  # Make wide matrix (rows = Lipid, columns = Sample)
  expr_matrix <- df %>%
    select(SampleID, Lipid, Value) %>%
    pivot_wider(names_from = SampleID, values_from = Value) %>%
    column_to_rownames("Lipid") %>%
    as.matrix()
  
  # Sample metadata
  sample_info <- df %>%
    distinct(SampleID, PairID, Timepoint) %>%
    arrange(SampleID)
  
  sample_info <- sample_info[match(colnames(expr_matrix), sample_info$SampleID), ]
  
  sample_info$Timepoint <- factor(sample_info$Timepoint,
                                  levels = c("T6","T24"))
  
  # Design matrix
  design <- model.matrix(~ 0 + Timepoint, data = sample_info)
  colnames(design) <- levels(sample_info$Timepoint)
  
  # Paired model
  corfit <- duplicateCorrelation(expr_matrix,
                                 design,
                                 block = sample_info$PairID)
  
  fit <- lmFit(expr_matrix,
               design,
               block = sample_info$PairID,
               correlation = corfit$consensus)
  
  contrast.matrix <- makeContrasts(
    T24vsT6 = T24 - T6,
    levels = design
  )
  
  fit2 <- contrasts.fit(fit, contrast.matrix)
  fit2 <- eBayes(fit2)
  
  results <- topTable(fit2,
                      coef = "T24vsT6",
                      number = Inf)
  
  results$Lipid <- rownames(results)
  results$Condition <- condition_name
  
  return(results)
}

prepare_lipid_long <- function(lipid_imputed, samples) {
  metadata_cols <- c("SampleID", "GroupName", "Batch", "PairID")
  
  lipid_imputed %>%
    filter(GroupName %in% samples) %>%
    mutate(across(-all_of(metadata_cols), ~ log2(.x + 1))) %>%
    pivot_longer(
      cols      = -all_of(metadata_cols),
      names_to  = "Lipid",
      values_to = "Value"
    ) %>%
    mutate(Timepoint = str_extract(GroupName, "T\\d+"))
}

run_paired_limma <- function(data, t_ref, t_comp) {
  
  df <- data %>%
    filter(Timepoint %in% c(t_ref, t_comp)) %>%
    group_by(PairID) %>%
    filter(n_distinct(Timepoint) == 2) %>%   
    ungroup()
  
  # Expression matrix: rows = Lipids, cols = Samples
  expr_matrix <- df %>%
    select(SampleID, Lipid, Value) %>%
    pivot_wider(names_from = SampleID, values_from = Value) %>%
    column_to_rownames("Lipid") %>%
    as.matrix()
  
  # Sample metadata aligned to columns
  sample_info <- df %>%
    distinct(SampleID, PairID, Timepoint)
  
  sample_info <- sample_info[
    match(colnames(expr_matrix), sample_info$SampleID),
  ]
  
  stopifnot(all(sample_info$SampleID == colnames(expr_matrix)))
  
  sample_info$Timepoint <- factor(sample_info$Timepoint,
                                  levels = c(t_ref, t_comp))
  
  # Design
  design <- model.matrix(~ Timepoint, data = sample_info)
  
  # Paired model
  corfit <- duplicateCorrelation(
    expr_matrix,
    design,
    block = sample_info$PairID
  )
  
  fit <- lmFit(
    expr_matrix,
    design,
    block = sample_info$PairID,
    correlation = corfit$consensus
  )
  
  fit <- eBayes(fit)
  
  results <- topTable(
    fit,
    coef = paste0("Timepoint", t_comp),
    number = Inf,
    adjust.method = "BH"
  ) %>%
    rownames_to_column("Lipid") %>%
    mutate(lipid_class = sub("\\..*", "", Lipid))
  
  results
}

results_NMP  <- run_paired_limma(lipid_long_NMP,  "T6",  "T24")
results_sNMP <- run_paired_limma(lipid_long_sNMP, "T3",  "T96")

plot_volcano <- function(results, title) {
  results %>%
    mutate(
      significant = case_when(
        adj.P.Val < 0.05 & abs(logFC) > 1 ~ "Significant",
        TRUE                             ~ "NS"
      )
    ) %>%
    ggplot(aes(logFC, -log10(adj.P.Val), color = lipid_class)) +
    geom_point(size = 4, alpha = 0.7) +
    geom_hline(yintercept = -log10(0.05), linetype = "dashed") +
    geom_vline(xintercept = c(-1, 1), linetype = "dashed") +
    scale_color_manual(values = class_colors, na.value = "grey70") +
    labs(
      title = title,
      x = "log2 fold‑change",
      y = "-log10 adjusted p‑value",
      color = "Lipid class"
    ) +
    theme_bw(base_size = 18, base_family = "Myriad Pro")
}
